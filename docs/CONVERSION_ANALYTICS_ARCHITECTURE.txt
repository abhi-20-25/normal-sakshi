         ┌──────────────────────────────────────────────────────────┐
         │  /api/analytics/footfall-conversion?days={N}             │
         │  ────────────────────────────────────────────────────────│
         │                                                           │
         │  STEP 1: Query Footfall Data                            │
         │    SELECT report_date, hour, SUM(in_count) as visitors  │
         │    FROM hourly_footfall                                  │
         │    WHERE report_date >= (today - N days)                │
         │    GROUP BY report_date, hour                           │
         │                                                           │
         │  STEP 2: Query Sales Data (JSONB Extraction)            │
         │    SELECT                                                │
         │      DATE(created_at AT TIME ZONE 'Asia/Kolkata'),      │
         │      EXTRACT(HOUR FROM created_at),                     │
         │      COUNT(DISTINCT orderID) as orders,                 │
         │      SUM(CAST(total AS DECIMAL)) as revenue             │
         │    FROM petpooja_webhook_events                         │
         │    WHERE date >= (today - N days)                       │
         │    GROUP BY date, hour                                  │
         │                                                           │
         │  STEP 3: Combine & Calculate Metrics                    │
         │    FOR EACH (date, hour):                               │
         │      • Lookup matching sales data                        │
         │      • Conversion Rate = (orders/visitors) * 100        │
         │      • Revenue per Visitor = revenue/visitors           │
         │      • Avg Order Value = revenue/orders                 │
         │                                                           │
         │  STEP 4: Aggregate Summary                              │
         │    • Total visitors, orders, revenue                     │
         │    • Overall conversion rate                             │
         │    • Overall revenue per visitor                         │
         │    • Overall average order value                         │
         │                                                           │
         └───────────────────────┬──────────────────────────────────┘

                                 │ JSON Response
